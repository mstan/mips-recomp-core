cmake_minimum_required(VERSION 3.20)
project(mips_recomp_core LANGUAGES CXX)
add_library(mips_recomp_core STATIC src/mips_decoder.cpp)
add_library(mips-recomp-core::core ALIAS mips_recomp_core)
target_include_directories(mips_recomp_core PUBLIC include)
target_compile_features(mips_recomp_core PRIVATE cxx_std_20 PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
 add_executable(mips_decoder_test tests/decoder_test.cpp)
 target_link_libraries(mips_decoder_test PRIVATE mips_recomp_core)
 target_compile_features(mips_decoder_test PRIVATE cxx_std_20)
 add_test(NAME decoder_contract COMMAND mips_decoder_test)
endif()

if(BUILD_TESTING)
 add_executable(mips_cxx17_consumer tests/cxx17_consumer.cpp)
 target_link_libraries(mips_cxx17_consumer PRIVATE mips_recomp_core)
 set_target_properties(mips_cxx17_consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
 add_test(NAME cxx17_consumer COMMAND mips_cxx17_consumer)
endif()
